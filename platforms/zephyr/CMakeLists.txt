if(CONFIG_POCKETJS)
  get_filename_component(POCKET_CHECKOUT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
  set(POCKET_SDK "" CACHE PATH "Exported PocketJS SDK")
  set(POCKET_PROFILE_EXPORT_DIR "" CACHE PATH "Directory from pocket host export")
  set(POCKET_RUNTIME_ARCHIVE "" CACHE FILEPATH "Product's single PocketJS Rust archive")
  if(NOT EXISTS "${POCKET_PROFILE_EXPORT_DIR}/pocket_profile.h")
    message(FATAL_ERROR "Export the product profile and set POCKET_PROFILE_EXPORT_DIR")
  endif()
  if(POCKET_SDK)
    # include/pocket holds the pocket_runtime.h that pocket_profile.h and
    # pocket/zephyr.h include.
    set(pocket_headers "${POCKET_SDK}/include" "${POCKET_SDK}/include/pocket")
  else()
    set(pocket_headers "${POCKET_CHECKOUT_ROOT}/contracts/generated")
  endif()
  if(POCKET_RUST_TARGET)
    set(pocket_triple "${POCKET_RUST_TARGET}")
  elseif(CONFIG_ARCH_POSIX)
    if(CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
      if(NOT CONFIG_64BIT)
        message(FATAL_ERROR "Set POCKET_RUST_TARGET for a 32-bit cross-host native simulator")
      endif()
      set(pocket_triple aarch64-unknown-linux-gnu)
    elseif(CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|i.86")
      if(CONFIG_64BIT)
        set(pocket_triple x86_64-unknown-linux-gnu)
      else()
        set(pocket_triple i686-unknown-linux-gnu)
      endif()
    else()
      message(FATAL_ERROR "Set POCKET_RUST_TARGET for this native simulator host")
    endif()
  elseif(CONFIG_CPU_CORTEX_M0 OR CONFIG_CPU_CORTEX_M0PLUS)
    set(pocket_triple thumbv6m-none-eabi)
  elseif(CONFIG_CPU_CORTEX_M3)
    set(pocket_triple thumbv7m-none-eabi)
  elseif(CONFIG_CPU_CORTEX_M4 OR CONFIG_CPU_CORTEX_M7)
    if(CONFIG_FPU)
      set(pocket_triple thumbv7em-none-eabihf)
    else()
      set(pocket_triple thumbv7em-none-eabi)
    endif()
  elseif(CONFIG_CPU_CORTEX_M33 OR CONFIG_CPU_CORTEX_M55)
    if(CONFIG_FPU)
      set(pocket_triple thumbv8m.main-none-eabihf)
    else()
      set(pocket_triple thumbv8m.main-none-eabi)
    endif()
  elseif(CONFIG_RISCV AND CONFIG_64BIT)
    set(pocket_triple riscv64gc-unknown-none-elf)
  elseif(CONFIG_RISCV)
    set(pocket_triple riscv32imac-unknown-none-elf)
  else()
    message(FATAL_ERROR "Declare POCKET_RUST_TARGET for this architecture")
  endif()
  set(POCKET_RUST_TARGET "${pocket_triple}" CACHE STRING "Rust target triple")
  if(CONFIG_POCKETJS_RUST_FROM_SOURCE)
    include(ExternalProject)
    if(NOT POCKET_PROGRAM_MANIFEST OR NOT POCKET_PROGRAM_ARCHIVE)
      message(FATAL_ERROR "Source builds require POCKET_PROGRAM_MANIFEST and POCKET_PROGRAM_ARCHIVE")
    endif()
    find_program(POCKET_CARGO cargo REQUIRED)
    set(pocket_target "${CMAKE_CURRENT_BINARY_DIR}/cargo")
    set(POCKET_RUNTIME_ARCHIVE "${pocket_target}/${POCKET_RUST_TARGET}/release/${POCKET_PROGRAM_ARCHIVE}")
    get_filename_component(pocket_program_dir "${POCKET_PROGRAM_MANIFEST}" DIRECTORY)
    set(POCKET_RUST_BUILD_STD "" CACHE STRING "Optional comma-separated standard library crates for tier-3 targets")
    set(pocket_cargo_flags)
    if(POCKET_RUST_BUILD_STD)
      list(APPEND pocket_cargo_flags "-Zbuild-std=${POCKET_RUST_BUILD_STD}")
    endif()
    ExternalProject_Add(pocket_program SOURCE_DIR "${pocket_program_dir}" CONFIGURE_COMMAND "" DOWNLOAD_COMMAND ""
      BUILD_COMMAND "${CMAKE_COMMAND}" -E env "CARGO_TARGET_DIR=${pocket_target}" "CARGO_PROFILE_RELEASE_PANIC=abort"
        "${POCKET_CARGO}" build ${pocket_cargo_flags} --locked --release --no-default-features --target "${POCKET_RUST_TARGET}" --manifest-path "${POCKET_PROGRAM_MANIFEST}"
      INSTALL_COMMAND "" BUILD_ALWAYS TRUE BUILD_BYPRODUCTS "${POCKET_RUNTIME_ARCHIVE}")
  elseif(NOT POCKET_RUNTIME_ARCHIVE)
    set(POCKET_RUNTIME_ARCHIVE "${POCKET_SDK}/lib/${POCKET_RUST_TARGET}/libpocket_runtime.a")
  endif()
  if(NOT CONFIG_POCKETJS_RUST_FROM_SOURCE AND NOT EXISTS "${POCKET_RUNTIME_ARCHIVE}")
    message(FATAL_ERROR "Missing PocketJS archive for ${POCKET_RUST_TARGET}")
  endif()
  zephyr_library_named(pocket_zephyr)
  zephyr_include_directories(${pocket_headers} "${CMAKE_CURRENT_LIST_DIR}/include" "${POCKET_PROFILE_EXPORT_DIR}")
  zephyr_library_sources(hooks/hooks.c)
  zephyr_library_sources_ifdef(CONFIG_POCKETJS_DISPLAY display/display.c)
  zephyr_library_sources_ifdef(CONFIG_POCKETJS_INPUT input/input.c)
  zephyr_link_libraries("${POCKET_RUNTIME_ARCHIVE}")
  if(CONFIG_POCKETJS_RUST_FROM_SOURCE)
    add_dependencies(pocket_zephyr pocket_program)
  endif()
  if(CONFIG_POCKETJS_GUEST_QUICKJS)
    if(NOT POCKET_QUICKJS_TARGET OR NOT POCKET_QUICKJS_INCLUDE_DIR)
      message(FATAL_ERROR "Supply the QuickJS module target/include directory")
    endif()
    if(NOT TARGET ${POCKET_QUICKJS_TARGET})
      message(FATAL_ERROR "Supply the QuickJS module target/include directory")
    endif()
    if(POCKET_SDK)
      set(guest_source "${POCKET_SDK}/src/guest-quickjs")
    else()
      set(guest_source "${POCKET_CHECKOUT_ROOT}/engine/guest-quickjs/src")
      zephyr_include_directories("${POCKET_CHECKOUT_ROOT}/engine/guest-quickjs/include")
    endif()
    zephyr_library_sources("${guest_source}/guest.c")
    zephyr_library_include_directories("${guest_source}")
    zephyr_include_directories("${POCKET_QUICKJS_INCLUDE_DIR}")
    zephyr_library_link_libraries(${POCKET_QUICKJS_TARGET})
  endif()
endif()
