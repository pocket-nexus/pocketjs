cmake_minimum_required(VERSION 3.16)
project(pocket_worker C)
add_library(pocket_worker STATIC src/worker.c)
target_include_directories(pocket_worker PUBLIC include)
target_compile_features(pocket_worker PUBLIC c_std_11)
target_compile_options(pocket_worker PRIVATE -Wall -Wextra -Werror)
if(DEFINED POCKET_QUICKJS_INCLUDE)
  target_sources(pocket_worker PRIVATE src/quickjs.c src/quickjs_ui.c src/runner.c)
  target_include_directories(pocket_worker PUBLIC "${POCKET_QUICKJS_INCLUDE}")
endif()
